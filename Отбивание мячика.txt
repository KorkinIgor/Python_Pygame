import pygame
import random

# Инициализация Pygame
pygame.init()

# Константы
WIDTH, HEIGHT = 800, 600
BALL_RADIUS = 15
PADDLE_WIDTH, PADDLE_HEIGHT = 100, 10
FPS = 60
a = 4

# Цвета
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
RED = (255, 0, 0)

# Создание окна
screen = pygame.display.set_mode((WIDTH, HEIGHT))
pygame.display.set_caption("Отбивание мячика")

# Класс для мяча
class Ball:
    def __init__(self):
        self.x = WIDTH // 2
        self.y = HEIGHT // 2
        self.radius = BALL_RADIUS
        self.dx = random.choice([-4, 4])
        self.dy = -a
        print(a)


    def move(self):
        self.x += self.dx
        self.y += self.dy

        # Проверка на столкновение с границами
        if self.x <= self.radius or self.x >= WIDTH - self.radius:
            self.dx *= -1
        if self.y <= self.radius:
            self.dy *= -1

    def draw(self, screen):
        pygame.draw.circle(screen, RED, (self.x, self.y), self.radius)

# Класс для ракетки
class Paddle:
    def __init__(self):
        self.x = WIDTH // 2 - PADDLE_WIDTH // 2
        self.y = HEIGHT - PADDLE_HEIGHT - 10
        self.width = PADDLE_WIDTH
        self.height = PADDLE_HEIGHT

    def move(self, dx):
        self.x += dx
        # Ограничение движения ракетки
        if self.x < 0:
            self.x = 0
        if self.x > WIDTH - self.width:
            self.x = WIDTH - self.width

    def draw(self, screen):
        pygame.draw.rect(screen, WHITE, (self.x, self.y, self.width, self.height))

# Основная функция игры
def main():
    clock = pygame.time.Clock()
    ball = Ball()
    paddle = Paddle()
    running = True

    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False

        keys = pygame.key.get_pressed()
        if keys[pygame.K_LEFT]:
            paddle.move(-8)
        if keys[pygame.K_RIGHT]:
            paddle.move(8)

        ball.move()

        # Проверка на столкновение с ракеткой
        if (paddle.x < ball.x < paddle.x + paddle.width) and (paddle.y < ball.y + ball.radius < paddle.y + paddle.height):
            ball.dy *= -1
            ball.y = paddle.y - ball.radius  # Убедитесь, что мяч не застревает в ракетке
            global a

        # Проверка на проигрыш
        if ball.y > HEIGHT:
            print("Игра окончена!")
            running = False

        # Отрисовка
        screen.fill(BLACK)
        ball.draw(screen)
        paddle.draw(screen)
        pygame.display.flip()
        clock.tick(FPS)

    pygame.quit()


main()
