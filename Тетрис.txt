import pygame
import random

# Константы
WIDTH, HEIGHT = 300, 600
BLOCK_SIZE = 30
GRID_WIDTH = WIDTH // BLOCK_SIZE
GRID_HEIGHT = HEIGHT // BLOCK_SIZE

# Цвета
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
COLORS = [
    (0, 255, 255),  # Циан
    (255, 165, 0),  # Оранжевый
    (0, 0, 255),  # Синий
    (255, 0, 0),  # Красный
    (128, 0, 128),  # Фиолетовый
    (0, 255, 0),  # Зеленый
    (255, 255, 0)  # Желтый
]

# Фигуры Тетромино
SHAPES = [
    [[1], [1], [1], [1]],  # I-образная фигура
    [[1, 1], [1, 1]],  # O-образная фигура
    [[0, 1], [1, 1], [1]],  # L-образная фигура
    [[1, 1], [1], [0]],  # J-образная фигура
    [[0, 1], [0, 1], [1]],  # S-образная фигура
    [[1], [1, 1], [0]],  # Z-образная фигура
    [[0, 1], [1, 1], [0]]  # T-образная фигура
]


class Tetris:
    def __init__(self):
        self.grid = [[BLACK for _ in range(GRID_WIDTH)] for _ in range(GRID_HEIGHT)]
        self.current_piece = self.new_piece()
        self.next_piece = self.new_piece()
        self.score = 0

    def new_piece(self):
        shape = random.choice(SHAPES)
        color = COLORS[SHAPES.index(shape)]
        return Piece(shape, color)

    def rotate_piece(self):
        self.current_piece.rotate()
        if self.check_collision():
            self.current_piece.rotate(-1)

    def check_collision(self):
        for y in range(len(self.current_piece.shape)):
            for x in range(len(self.current_piece.shape[y])):
                if self.current_piece.shape[y][x]:
                    grid_x = self.current_piece.x + x
                    grid_y = self.current_piece.y + y

                    if grid_x < 0 or grid_x >= GRID_WIDTH or grid_y >= GRID_HEIGHT:
                        return True

                    if grid_y >= 0 and self.grid[grid_y][grid_x] != BLACK:
                        return True

        return False

    def merge_piece(self):
        for y in range(len(self.current_piece.shape)):
            for x in range(len(self.current_piece.shape[y])):
                if self.current_piece.shape[y][x]:
                    grid_x = self.current_piece.x + x
                    grid_y = self.current_piece.y + y

                    if grid_y >= 0:
                        self.grid[grid_y][grid_x] = self.current_piece.color

    def clear_lines(self):
        lines_to_clear = []

        for i in range(GRID_HEIGHT):
            if all(self.grid[i][j] != BLACK for j in range(GRID_WIDTH)):
                lines_to_clear.append(i)

        for i in lines_to_clear:
            del self.grid[i]
            self.grid.insert(0, [BLACK for _ in range(GRID_WIDTH)])
            self.score += len(lines_to_clear)


class Piece:
    def __init__(self, shape, color):
        self.shape = shape
        self.color = color
        self.x = GRID_WIDTH // 2 - len(shape[0]) // 2
        self.y = -len(shape)

    def rotate(self):
        self.shape = list(zip(*self.shape[::-1]))


def draw_grid(screen):
    for y in range(GRID_HEIGHT):
        for x in range(GRID_WIDTH):
            pygame.draw.rect(screen,
                             Tetris().grid[y][x],
                             (x * BLOCK_SIZE,
                              y * BLOCK_SIZE,
                              BLOCK_SIZE,
                              BLOCK_SIZE),
                             border_radius=3)
            pygame.draw.rect(screen,
                             WHITE,
                             (x * BLOCK_SIZE,
                              y * BLOCK_SIZE,
                              BLOCK_SIZE,
                              BLOCK_SIZE),
                             width=2,
                             border_radius=3)


def main():
    pygame.init()

    screen = pygame.display.set_mode((WIDTH, HEIGHT))

    clock = pygame.time.Clock()

    game_over = False

    tetris_game = Tetris()

    while not game_over:
        screen.fill(BLACK)

        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                game_over = True

            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_LEFT:
                    tetris_game.current_piece.x -= 1
                    if tetris_game.check_collision():
                        tetris_game.current_piece.x += 1

                elif event.key == pygame.K_RIGHT:
                    tetris_game.current_piece.x += 1
                    if tetris_game.check_collision():
                        tetris_game.current_piece.x -= 1

                elif event.key == pygame.K_DOWN:
                    tetris_game.current_piece.y += 1
                    if tetris_game.check_collision():
                        tetris_game.current_piece.y -= 1

                elif event.key == pygame.K_UP:
                    tetris_game.rotate_piece()

        tetris_game.current_piece.y += 1

        if tetris_game.check_collision():
            tetris_game.current_piece.y -= 1
            tetris_game.merge_piece()
            tetris_game.clear_lines()
            tetris_game.current_piece = tetris_game.next_piece
            tetris_game.next_piece = tetris_game.new_piece()

            if tetris_game.check_collision():
                game_over = True

        draw_grid(screen)

        for y in range(GRID_HEIGHT):
            for x in range(GRID_WIDTH):
                if tetris_game.grid[y][x] != BLACK:
                    pygame.draw.rect(screen,
                                     tetris_game.grid[y][x],
                                     (x * BLOCK_SIZE,
                                      y * BLOCK_SIZE,
                                      BLOCK_SIZE,
                                      BLOCK_SIZE))

        piece_shape = tetris_game.current_piece.shape

        for y in range(len(piece_shape)):
            for x in range(len(piece_shape[y])):
                if piece_shape[y][x]:
                    pygame.draw.rect(screen,
                                     tetris_game.current_piece.color,
                                     ((tetris_game.current_piece.x + x) * BLOCK_SIZE,
                                      (tetris_game.current_piece.y + y) * BLOCK_SIZE,
                                      BLOCK_SIZE,
                                      BLOCK_SIZE))

        pygame.display.flip()

        clock.tick(10)


if __name__ == "__main__":
    main()
